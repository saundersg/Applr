mpg <- ggplot2::mpg

raises("band_not_numeric", function() {
  ggplot2::ggplot(mpg, ggplot2::aes(displ, hwy)) + geom_slice(lm(hwy ~ displ + drv, mpg), band = "drv")
}, c("warning", "message"), "is not numeric, so it has no range",
