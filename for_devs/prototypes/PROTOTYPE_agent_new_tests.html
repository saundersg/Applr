autoplot.data.frame <- function(object, columns = NULL, ...) {
  num <- names(object)[vapply(object, is.numeric, TRUE)]
  dropped <- setdiff(columns %||% names(object), num)
  if (length(dropped)) message("Dropping non-numeric columns: ", toString(dropped))
  keep <- intersect(columns %||% names(object), num)
  if (length(keep) < 2) stop("A pairs plot needs at least two numeric columns.")
  # one panel per pair of columns, a density on the diagonal
  ...
}
